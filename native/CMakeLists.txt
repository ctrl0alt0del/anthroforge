cmake_minimum_required(VERSION 3.20)
project(anthroforge_native LANGUAGES CXX)
enable_testing()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_compile_definitions(NOMINMAX EIGEN_DONT_PARALLELIZE EIGEN_MPL2_ONLY)
if(NOT MSVC)
  add_compile_options(-O2 -ffp-contract=off)
endif()
add_subdirectory(smoothing)
find_package(OpenMP REQUIRED)
add_library(af_fiber_ccd_v3_57 SHARED fiber_ccd/af_fiber_ccd.cpp)
target_compile_definitions(af_fiber_ccd_v3_57 PRIVATE AF_V3_NATIVE_CONTROLLER=1)
target_link_libraries(af_fiber_ccd_v3_57 PRIVATE af_smoothing_core OpenMP::OpenMP_CXX)
add_library(af_deformation_v1 SHARED deformation/native.cpp)
target_include_directories(af_deformation_v1 PRIVATE deformation/vendor smoothing/vendor)
add_library(af_topology_field_v1 SHARED topology/field.cpp)
target_include_directories(af_topology_field_v1 PRIVATE smoothing/vendor)
install(TARGETS af_fiber_ccd_v3_57 LIBRARY DESTINATION fiber_ccd)
install(TARGETS af_deformation_v1 LIBRARY DESTINATION deformation)
install(TARGETS af_topology_field_v1 LIBRARY DESTINATION topology)
install(TARGETS af_height_smoothing_v1 LIBRARY DESTINATION smoothing)
