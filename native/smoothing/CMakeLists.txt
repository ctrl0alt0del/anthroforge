cmake_minimum_required(VERSION 3.20)
project(af_height_smoothing LANGUAGES CXX)
add_library(af_smoothing_core STATIC src/core.cpp)
set_target_properties(af_smoothing_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_features(af_smoothing_core PUBLIC cxx_std_17)
target_include_directories(af_smoothing_core PUBLIC include vendor)
target_compile_definitions(af_smoothing_core PUBLIC NOMINMAX EIGEN_DONT_PARALLELIZE EIGEN_MPL2_ONLY)
add_library(af_height_smoothing_v1 SHARED src/height.cpp)
target_link_libraries(af_height_smoothing_v1 PRIVATE af_smoothing_core)
enable_testing()
add_executable(af_smoothing_tests tests/core_tests.cpp)
target_link_libraries(af_smoothing_tests PRIVATE af_smoothing_core)
add_test(NAME smoothing_core COMMAND af_smoothing_tests)
